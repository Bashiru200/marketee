const PAGE_SIZE = 12

// Only select columns that exist in the current businesses schema
const BUSINESS_FIELDS = `
  id,
  name,
  category,
  description,
  street,
  address,
  city,
  state,
  zip,
  country,
  phone,
  website,
  cover_image,
  logo_url,
  rating,
  review_count,
  tags,
  lat,
  lng,
  verified,
  premium,
  featured,
  plan
`

const CATEGORIES = [
  { id: 'food',       name: 'Food & Groceries', icon: '🍲' },
  { id: 'restaurant', name: 'Restaurants',       icon: '🍽️' },
  { id: 'fashion',    name: 'Fashion & Fabric',  icon: '👗' },
  { id: 'beauty',     name: 'Beauty & Hair',     icon: '💇' },
  { id: 'herbs',      name: 'Herbs & Wellness',  icon: '🌿' },
  { id: 'music',      name: 'Music & Arts',      icon: '🎵' },
  { id: 'crafts',     name: 'Crafts & Decor',    icon: '🏺' },
  { id: 'services',   name: 'Services',          icon: '🛠️' },
  { id: 'nightlife',  name: 'Bars & Nightlife',  icon: '🍹' },
] as const

type BusinessCategory = (typeof CATEGORIES)[number]

export interface SearchBusiness {
  id:            string
  name:          string
  category:      string | null
  description:   string | null
  street:        string | null
  address:       string | null
  city:          string | null
  state:         string | null
  zip:           string | null
  country:       string | null
  phone:         string | null
  website:       string | null
  cover_image:   string | null
  logo_url:      string | null
  rating:        number | null
  review_count:  number | null
  tags:          string[] | null
  lat:           number | null
  lng:           number | null
  verified:      boolean
  premium:       boolean
  featured:      boolean
  plan:          string | null
}

export interface SearchCategory extends BusinessCategory {
  count: number
}

async function getInitialSearchData(): Promise<{
  businesses:    SearchBusiness[]
  totalCount:    number
  categories:    SearchCategory[]
  errorMessage:  string | null
}> {
  const supabase = await createClient()

  // ── Businesses ───────────────────────────────────────────────────────
  const { data, count, error } = await supabase
    .from('businesses')
    .select(BUSINESS_FIELDS, { count: 'exact' })
    .order('featured', { ascending: false, nullsFirst: false })
    .order('verified', { ascending: false, nullsFirst: false })
    .order('rating',   { ascending: false, nullsFirst: false })
    .order('created_at', { ascending: false })
    .range(0, PAGE_SIZE - 1)

  if (error) {
    console.error('[search-page] Initial business query failed:', {
      message: error.message,
      code:    error.code,
      details: error.details,
      hint:    error.hint,
    })
    return {
      businesses:   [],
      totalCount:   0,
      categories:   await buildCategoriesWithCounts(supabase),
      errorMessage: 'Businesses could not be loaded. Please refresh the page and try again.',
    }
  }

  return {
    businesses:   normalizeBusinesses(data),
    totalCount:   count ?? 0,
    categories:   await buildCategoriesWithCounts(supabase),
    errorMessage: null,
  }
}

/**
 * Build the categories list with real per-category counts from the DB.
 * Uses a lightweight SELECT category, count() grouping.
 */
async function buildCategoriesWithCounts(
  supabase: Awaited<ReturnType<typeof createClient>>,
): Promise<SearchCategory[]> {
  try {
    const results = await Promise.all(
      CATEGORIES.map(async cat => {
        const { count } = await supabase
          .from('businesses')
          .select('id', { count: 'exact', head: true })
          .eq('category', cat.id)
        return { ...cat, count: count ?? 0 }
      }),
    )
    return results
  } catch (err) {
    console.error('[search-page] Category counts failed:', err)
    return CATEGORIES.map(cat => ({ ...cat, count: 0 }))
  }
}

function normalizeBusinesses(businesses: unknown[] | null): SearchBusiness[] {
  if (!Array.isArray(businesses)) return []

  return businesses.map(item => {
    const b = item as Partial<SearchBusiness>
    return {
      id:            String(b.id ?? ''),
      name:          b.name?.trim() || 'Unnamed business',
      category:      b.category      ?? null,
      description:   b.description   ?? null,
      street:        b.street        ?? null,
      address:       b.address       ?? null,
      city:          b.city          ?? null,
      state:         b.state         ?? null,
      zip:           b.zip           ?? null,
      country:       b.country       ?? null,
      phone:         b.phone         ?? null,
      website:       b.website       ?? null,
      cover_image:   b.cover_image   ?? null,
      logo_url:      b.logo_url      ?? null,
      rating:        typeof b.rating       === 'number' ? b.rating       : 0,
      review_count:  typeof b.review_count === 'number' ? b.review_count : 0,
      tags:          Array.isArray(b.tags) ? b.tags : null,
      lat:           typeof b.lat === 'number' ? b.lat : null,
      lng:           typeof b.lng === 'number' ? b.lng : null,
      verified:      Boolean(b.verified),
      premium:       Boolean(b.premium),
      featured:      Boolean(b.featured),
      plan:          b.plan ?? null,
    }
  })
}

async function SearchContent() {
  const { businesses, totalCount, categories, errorMessage } = await getInitialSearchData()

  return (
    <SearchClient
      initialBusinesses={businesses}
      totalCount={totalCount}
      pageSize={PAGE_SIZE}
      categories={categories}
      initialTab="businesses"
      initialError={errorMessage}
    />
  )
}

export default function SearchPage() {
  return (
    <main className="min-h-screen bg-[#F7FAF8]">
      <Suspense fallback={<SearchLoading />}>
        <SearchContent />
      </Suspense>
    </main>
  )
}
