'use client'

import * as React from 'react'
import Link from 'next/link'
import {
  ArrowRight,
  BriefcaseBusiness,
  Building2,
  Check,
  ChevronDown,
  Eye,
  EyeOff,
  Globe2,
  Heart,
  LockKeyhole,
  Mail,
  MapPin,
  Phone,
  Store,
  Tag,
  TrendingUp,
  Upload,
  UserRound,
  UsersRound,
} from 'lucide-react'

type LocationType = 'physical' | 'online' | 'both'

interface BusinessFormData {
  businessName: string
  category: string
  locationType: LocationType

  country: string
  city: string
  state: string
  zipCode: string
  address: string
  addressLine2: string

  phone: string
  businessEmail: string
  website: string
  onlineOrderingUrl: string
  serviceArea: string

  description: string

  ownerName: string
  ownerRole: string
  accountEmail: string
  password: string
  confirmPassword: string

  agreed: boolean
}

const INITIAL_FORM: BusinessFormData = {
  businessName: '',
  category: '',
  locationType: 'physical',

  country: 'United States',
  city: '',
  state: '',
  zipCode: '',
  address: '',
  addressLine2: '',

  phone: '',
  businessEmail: '',
  website: '',
  onlineOrderingUrl: '',
  serviceArea: '',

  description: '',

  ownerName: '',
  ownerRole: 'Owner',
  accountEmail: '',
  password: '',
  confirmPassword: '',

  agreed: false,
}

const CATEGORIES = [
  'Food & Groceries',
  'Restaurants',
  'Fashion & Fabric',
  'Beauty & Hair',
  'Herbs & Wellness',
  'Music & Arts',
  'Crafts & Decor',
  'Professional Services',
  'Bars & Nightlife',
  'Other',
]

const BRAND = {
  dark: '#053528',
  primary: '#085041',
  green: '#1D9E75',
  mint: '#E1F5EE',
  lightMint: '#F4FBF8',
}

export default function CreateBusinessAccountPage() {
  const [form, setForm] =
    React.useState<BusinessFormData>(INITIAL_FORM)

  const [showPassword, setShowPassword] =
    React.useState(false)

  const [showConfirmPassword, setShowConfirmPassword] =
    React.useState(false)

  const [logo, setLogo] = React.useState<File | null>(null)
  const [submitting, setSubmitting] =
    React.useState(false)

  const [error, setError] = React.useState('')
  const [success, setSuccess] = React.useState(false)

  const hasPhysicalLocation =
    form.locationType === 'physical' ||
    form.locationType === 'both'

  const hasOnlineLocation =
    form.locationType === 'online' ||
    form.locationType === 'both'

  function updateField<K extends keyof BusinessFormData>(
    field: K,
    value: BusinessFormData[K],
  ) {
    setForm((current) => ({
      ...current,
      [field]: value,
    }))

    setError('')
  }

  function selectLocationType(type: LocationType) {
    setForm((current) => ({
      ...current,
      locationType: type,
    }))
  }

  function validateForm() {
    if (!form.businessName.trim()) {
      return 'Enter your business name.'
    }

    if (!form.category) {
      return 'Select a business category.'
    }

    if (hasPhysicalLocation) {
      if (!form.address.trim()) {
        return 'Enter the physical business address.'
      }

      if (!form.city.trim()) {
        return 'Enter the business city.'
      }

      if (!form.state) {
        return 'Select the business state.'
      }

      if (!form.zipCode.trim()) {
        return 'Enter the business ZIP code.'
      }
    }

    if (hasOnlineLocation && !form.website.trim()) {
      return 'Enter the website for your online business.'
    }

    if (!form.phone.trim()) {
      return 'Enter the business phone number.'
    }

    if (!form.businessEmail.trim()) {
      return 'Enter the public business email.'
    }

    if (form.description.trim().length < 40) {
      return 'Write a business description containing at least 40 characters.'
    }

    if (!form.ownerName.trim()) {
      return 'Enter the account owner’s full name.'
    }

    if (!form.accountEmail.trim()) {
      return 'Enter the email you will use to sign in.'
    }

    if (form.password.length < 8) {
      return 'Your password must contain at least 8 characters.'
    }

    if (form.password !== form.confirmPassword) {
      return 'The passwords do not match.'
    }

    if (!form.agreed) {
      return 'You must agree to the Terms of Service and Privacy Policy.'
    }

    return null
  }

  async function handleSubmit(
    event: React.FormEvent<HTMLFormElement>,
  ) {
    event.preventDefault()

    const validationError = validateForm()

    if (validationError) {
      setError(validationError)
      return
    }

    setSubmitting(true)
    setError('')

    try {
      const payload = new FormData()

      Object.entries(form).forEach(([key, value]) => {
        payload.append(key, String(value))
      })

      if (logo) {
        payload.append('logo', logo)
      }

      const response = await fetch(
        '/api/businesses/create-account',
        {
          method: 'POST',
          body: payload,
        },
      )

      const result = await response.json().catch(() => null)

      if (!response.ok) {
        throw new Error(
          result?.error ??
            'Your business account could not be created.',
        )
      }

      setSuccess(true)
    } catch (submitError) {
      setError(
        submitError instanceof Error
          ? submitError.message
          : 'Your business account could not be created.',
      )
    } finally {
      setSubmitting(false)
    }
  }

  if (success) {
    return (
      <main className="flex min-h-screen items-center justify-center bg-gray-50 px-4 py-12">
        <div className="w-full max-w-xl rounded-3xl border border-gray-200 bg-white p-8 text-center shadow-sm sm:p-12">
          <div
            className="mx-auto flex h-20 w-20 items-center justify-center rounded-full"
            style={{ backgroundColor: BRAND.mint }}
          >
            <Check
              size={34}
              style={{ color: BRAND.green }}
            />
          </div>

          <h1 className="mt-6 text-2xl font-extrabold text-gray-950">
            Your business account is ready
          </h1>

          <p className="mx-auto mt-3 max-w-md text-sm leading-7 text-gray-500">
            We created your Markeetee business account.
            Check your email to confirm your address and
            continue setting up your listing.
          </p>

          <Link
            href="/dashboard"
            className="mt-8 inline-flex items-center gap-2 rounded-xl px-6 py-3 text-sm font-bold text-white"
            style={{ backgroundColor: BRAND.primary }}
          >
            Continue to dashboard
            <ArrowRight size={16} />
          </Link>
        </div>
      </main>
    )
  }

  return (
    <main className="min-h-screen bg-gray-50">
      <div className="mx-auto grid min-h-screen max-w-[1500px] grid-cols-1 lg:grid-cols-[390px_minmax(0,1fr)]">
        <BusinessBenefitsPanel />

        <section className="px-4 py-6 sm:px-8 sm:py-10 lg:px-10">
          <form
            onSubmit={handleSubmit}
            className="mx-auto max-w-4xl overflow-hidden rounded-3xl border border-gray-200 bg-white shadow-sm"
          >
            <div className="border-b border-gray-100 px-6 py-7 sm:px-9">
              <div
                className="mb-5 flex h-12 w-12 items-center justify-center rounded-2xl lg:hidden"
                style={{ backgroundColor: BRAND.mint }}
              >
                <Store
                  size={22}
                  style={{ color: BRAND.primary }}
                />
              </div>

              <h1 className="text-2xl font-extrabold text-gray-950">
                Create your business account
              </h1>

              <p className="mt-2 text-sm leading-6 text-gray-500">
                Tell customers about your business and
                create the account you will use to manage
                your listing.
              </p>
            </div>

            <div className="space-y-9 px-6 py-8 sm:px-9">
              {error ? (
                <div className="rounded-2xl border border-red-200 bg-red-50 px-4 py-3 text-sm text-red-700">
                  {error}
                </div>
              ) : null}

              <FormSection
                icon={Building2}
                title="Business information"
                description="Add the basic details customers will see on your listing."
              >
                <FormField
                  label="Business name"
                  required
                  className="sm:col-span-2"
                >
                  <TextInput
                    value={form.businessName}
                    onChange={(value) =>
                      updateField('businessName', value)
                    }
                    placeholder="e.g. Mama Titi African Kitchen"
                    icon={Store}
                    required
                  />
                </FormField>

                <FormField
                  label="Business category"
                  required
                  className="sm:col-span-2"
                >
                  <SelectInput
                    value={form.category}
                    onChange={(value) =>
                      updateField('category', value)
                    }
                    placeholder="Select a category"
                    options={CATEGORIES}
                    icon={Tag}
                    required
                  />
                </FormField>

                <FormField
                  label="Business logo"
                  hint="Optional"
                  className="sm:col-span-2"
                >
                  <label className="flex cursor-pointer items-center gap-4 rounded-2xl border border-dashed border-gray-300 bg-gray-50 p-4 transition hover:border-emerald-400 hover:bg-emerald-50/40">
                    <div
                      className="flex h-11 w-11 shrink-0 items-center justify-center rounded-xl"
                      style={{ backgroundColor: BRAND.mint }}
                    >
                      <Upload
                        size={18}
                        style={{ color: BRAND.primary }}
                      />
                    </div>

                    <div className="min-w-0 flex-1">
                      <p className="truncate text-sm font-bold text-gray-800">
                        {logo
                          ? logo.name
                          : 'Upload your business logo'}
                      </p>

                      <p className="mt-1 text-xs text-gray-400">
                        PNG, JPG or WebP. Maximum 5 MB.
                      </p>
                    </div>

                    <input
                      type="file"
                      accept="image/png,image/jpeg,image/webp"
                      className="hidden"
                      onChange={(event) =>
                        setLogo(
                          event.target.files?.[0] ?? null,
                        )
                      }
                    />
                  </label>
                </FormField>

                <FormField
                  label="Short business description"
                  required
                  className="sm:col-span-2"
                >
                  <textarea
                    value={form.description}
                    onChange={(event) =>
                      updateField(
                        'description',
                        event.target.value,
                      )
                    }
                    maxLength={500}
                    rows={5}
                    required
                    placeholder="Describe your business, products, services, and what makes you different..."
                    className="w-full resize-none rounded-xl border border-gray-200 px-4 py-3 text-sm leading-7 text-gray-900 outline-none transition placeholder:text-gray-400 focus:border-emerald-500 focus:ring-4 focus:ring-emerald-500/10"
                  />

                  <p className="mt-1.5 text-right text-xs text-gray-400">
                    {form.description.length}/500
                  </p>
                </FormField>
              </FormSection>

              <FormSection
                icon={MapPin}
                title="Business location"
                description="Choose how customers can access your business."
              >
                <div className="grid gap-3 sm:col-span-2 sm:grid-cols-3">
                  <LocationOption
                    selected={
                      form.locationType === 'physical'
                    }
                    icon={Store}
                    title="Physical"
                    description="Customers visit your address"
                    onClick={() =>
                      selectLocationType('physical')
                    }
                  />

                  <LocationOption
                    selected={
                      form.locationType === 'online'
                    }
                    icon={Globe2}
                    title="Online"
                    description="Your business operates online"
                    onClick={() =>
                      selectLocationType('online')
                    }
                  />

                  <LocationOption
                    selected={form.locationType === 'both'}
                    icon={BriefcaseBusiness}
                    title="Both"
                    description="Online and physical location"
                    onClick={() =>
                      selectLocationType('both')
                    }
                  />
                </div>

                {hasPhysicalLocation ? (
                  <>
                    <FormField
                      label="Country"
                      required
                    >
                      <SelectInput
                        value={form.country}
                        onChange={(value) =>
                          updateField('country', value)
                        }
                        options={[
                          'United States',
                          'Canada',
                          'United Kingdom',
                          'Other',
                        ]}
                        icon={Globe2}
                        required
                      />
                    </FormField>

                    <FormField label="City" required>
                      <TextInput
                        value={form.city}
                        onChange={(value) =>
                          updateField('city', value)
                        }
                        placeholder="e.g. Houston"
                        required
                      />
                    </FormField>

                    <FormField label="State" required>
                      <SelectInput
                        value={form.state}
                        onChange={(value) =>
                          updateField('state', value)
                        }
                        options={US_STATES}
                        placeholder="Select state"
                        required
                      />
                    </FormField>

                    <FormField label="ZIP code" required>
                      <TextInput
                        value={form.zipCode}
                        onChange={(value) =>
                          updateField('zipCode', value)
                        }
                        placeholder="e.g. 77001"
                        inputMode="numeric"
                        required
                      />
                    </FormField>

                    <FormField
                      label="Business address"
                      required
                      className="sm:col-span-2"
                    >
                      <TextInput
                        value={form.address}
                        onChange={(value) =>
                          updateField('address', value)
                        }
                        placeholder="Street address"
                        icon={MapPin}
                        required
                      />
                    </FormField>

                    <FormField
                      label="Address line 2"
                      hint="Optional"
                      className="sm:col-span-2"
                    >
                      <TextInput
                        value={form.addressLine2}
                        onChange={(value) =>
                          updateField(
                            'addressLine2',
                            value,
                          )
                        }
                        placeholder="Suite, unit, floor or building"
                      />
                    </FormField>
                  </>
                ) : null}

                {hasOnlineLocation ? (
                  <div
                    className="sm:col-span-2 rounded-2xl border p-5"
                    style={{
                      borderColor: '#B7E4D5',
                      backgroundColor: BRAND.lightMint,
                    }}
                  >
                    <div className="mb-5 flex items-start gap-3">
                      <div
                        className="flex h-10 w-10 shrink-0 items-center justify-center rounded-xl"
                        style={{
                          backgroundColor: BRAND.mint,
                        }}
                      >
                        <Globe2
                          size={18}
                          style={{
                            color: BRAND.primary,
                          }}
                        />
                      </div>

                      <div>
                        <p className="text-sm font-bold text-gray-900">
                          Online business details
                        </p>

                        <p className="mt-1 text-xs leading-5 text-gray-500">
                          Help customers understand where
                          you deliver or provide services.
                        </p>
                      </div>
                    </div>

                    <div className="grid gap-5 sm:grid-cols-2">
                      <FormField
                        label="Business website"
                        required
                        className="sm:col-span-2"
                      >
                        <TextInput
                          type="url"
                          value={form.website}
                          onChange={(value) =>
                            updateField('website', value)
                          }
                          placeholder="https://yourbusiness.com"
                          icon={Globe2}
                          required
                        />
                      </FormField>

                      <FormField
                        label="Online ordering or booking URL"
                        hint="Optional"
                      >
                        <TextInput
                          type="url"
                          value={form.onlineOrderingUrl}
                          onChange={(value) =>
                            updateField(
                              'onlineOrderingUrl',
                              value,
                            )
                          }
                          placeholder="https://..."
                        />
                      </FormField>

                      <FormField
                        label="Service or delivery area"
                        hint="Optional"
                      >
                        <TextInput
                          value={form.serviceArea}
                          onChange={(value) =>
                            updateField(
                              'serviceArea',
                              value,
                            )
                          }
                          placeholder="e.g. Nationwide or Houston area"
                        />
                      </FormField>
                    </div>
                  </div>
                ) : null}
              </FormSection>

              <FormSection
                icon={Phone}
                title="Contact information"
                description="Customers may use these details to contact your business."
              >
                <FormField
                  label="Business phone"
                  required
                >
                  <TextInput
                    type="tel"
                    value={form.phone}
                    onChange={(value) =>
                      updateField('phone', value)
                    }
                    placeholder="(123) 456-7890"
                    icon={Phone}
                    required
                  />
                </FormField>

                <FormField
                  label="Business email"
                  required
                >
                  <TextInput
                    type="email"
                    value={form.businessEmail}
                    onChange={(value) =>
                      updateField(
                        'businessEmail',
                        value,
                      )
                    }
                    placeholder="hello@yourbusiness.com"
                    icon={Mail}
                    required
                  />
                </FormField>

                {!hasOnlineLocation ? (
                  <FormField
                    label="Business website"
                    hint="Optional"
                    className="sm:col-span-2"
                  >
                    <TextInput
                      type="url"
                      value={form.website}
                      onChange={(value) =>
                        updateField('website', value)
                      }
                      placeholder="https://yourbusiness.com"
                      icon={Globe2}
                    />
                  </FormField>
                ) : null}
              </FormSection>

              <FormSection
                icon={UserRound}
                title="Account owner"
                description="Create the account that will manage this business."
              >
                <FormField
                  label="Full name"
                  required
                >
                  <TextInput
                    value={form.ownerName}
                    onChange={(value) =>
                      updateField('ownerName', value)
                    }
                    placeholder="Your full name"
                    icon={UserRound}
                    autoComplete="name"
                    required
                  />
                </FormField>

                <FormField
                  label="Role in business"
                  required
                >
                  <SelectInput
                    value={form.ownerRole}
                    onChange={(value) =>
                      updateField('ownerRole', value)
                    }
                    options={[
                      'Owner',
                      'Co-owner',
                      'Manager',
                      'Marketing manager',
                      'Employee',
                      'Authorized representative',
                    ]}
                    icon={UsersRound}
                    required
                  />
                </FormField>

                <FormField
                  label="Account email"
                  required
                  className="sm:col-span-2"
                >
                  <TextInput
                    type="email"
                    value={form.accountEmail}
                    onChange={(value) =>
                      updateField(
                        'accountEmail',
                        value,
                      )
                    }
                    placeholder="The email you will use to sign in"
                    icon={Mail}
                    autoComplete="email"
                    required
                  />
                </FormField>

                <FormField label="Password" required>
                  <PasswordInput
                    value={form.password}
                    onChange={(value) =>
                      updateField('password', value)
                    }
                    visible={showPassword}
                    onToggle={() =>
                      setShowPassword((current) => !current)
                    }
                    placeholder="At least 8 characters"
                    autoComplete="new-password"
                  />
                </FormField>

                <FormField
                  label="Confirm password"
                  required
                >
                  <PasswordInput
                    value={form.confirmPassword}
                    onChange={(value) =>
                      updateField(
                        'confirmPassword',
                        value,
                      )
                    }
                    visible={showConfirmPassword}
                    onToggle={() =>
                      setShowConfirmPassword(
                        (current) => !current,
                      )
                    }
                    placeholder="Enter the password again"
                    autoComplete="new-password"
                  />
                </FormField>
              </FormSection>

              <div className="border-t border-gray-100 pt-7">
                <label className="flex cursor-pointer items-start gap-3">
                  <input
                    type="checkbox"
                    checked={form.agreed}
                    onChange={(event) =>
                      updateField(
                        'agreed',
                        event.target.checked,
                      )
                    }
                    className="mt-1 h-4 w-4 shrink-0 cursor-pointer accent-emerald-700"
                  />

                  <span className="text-sm leading-6 text-gray-600">
                    I agree to Markeetee&apos;s{' '}
                    <Link
                      href="/terms"
                      className="font-semibold hover:underline"
                      style={{ color: BRAND.primary }}
                    >
                      Terms of Service
                    </Link>{' '}
                    and{' '}
                    <Link
                      href="/privacy"
                      className="font-semibold hover:underline"
                      style={{ color: BRAND.primary }}
                    >
                      Privacy Policy
                    </Link>
                    .
                  </span>
                </label>

                <button
                  type="submit"
                  disabled={submitting}
                  className="mt-6 flex w-full items-center justify-center gap-2 rounded-xl px-6 py-3.5 text-sm font-bold text-white transition hover:opacity-95 disabled:cursor-not-allowed disabled:opacity-60"
                  style={{
                    backgroundColor: BRAND.primary,
                  }}
                >
                  {submitting
                    ? 'Creating business account...'
                    : 'Create business account'}

                  {!submitting ? (
                    <ArrowRight size={16} />
                  ) : null}
                </button>

                <p className="mt-4 text-center text-xs leading-5 text-gray-400">
                  Already have an account?{' '}
                  <Link
                    href="/auth/login"
                    className="font-bold hover:underline"
                    style={{ color: BRAND.primary }}
                  >
                    Sign in
                  </Link>
                </p>
              </div>
            </div>
          </form>
        </section>
      </div>
    </main>
  )
}

                  {benefit.description}
                </p>
              </div>
            </div>
          )

function TextInput({

  onChange: (value: string) => void

function SelectInput({
  value,
  onChange,
  options,
  placeholder,
  icon: Icon,
  required,
}: {
  value: string
  onChange: (value: string) => void
  options: string[]
  placeholder?: string
  icon?: React.ElementType
  required?: boolean
}) {
  return (
    <div className="relative">
      {Icon ? (
        <Icon
          size={17}
          className="pointer-events-none absolute left-4 top-1/2 z-10 -translate-y-1/2 text-gray-400"
        />
      ) : null}

      <select
        value={value}
        onChange={(event) =>
          onChange(event.target.value)
        }
        required={required}
        className={`w-full appearance-none rounded-xl border border-gray-200 bg-white py-3 text-sm text-gray-900 outline-none transition focus:border-emerald-500 focus:ring-4 focus:ring-emerald-500/10 ${
          Icon ? 'pl-11 pr-10' : 'pl-4 pr-10'
        }`}
      >
        {placeholder ? (
          <option value="">{placeholder}</option>
        ) : null}

        {options.map((option) => (
          <option key={option} value={option}>
            {option}
          </option>
        ))}
      </select>

      <ChevronDown
        size={17}
        className="pointer-events-none absolute right-4 top-1/2 -translate-y-1/2 text-gray-400"
      />
    </div>
  )
}

function PasswordInput({
  value,
  onChange,
  visible,
  onToggle,
  placeholder,
  autoComplete,
}: {
  value: string
  onChange: (value: string) => void
  visible: boolean
  onToggle: () => void
  placeholder?: string
  autoComplete?: string
}) {
  return (
    <div className="relative">
      <LockKeyhole
        size={17}
        className="pointer-events-none absolute left-4 top-1/2 -translate-y-1/2 text-gray-400"
      />

      <input
        type={visible ? 'text' : 'password'}
        value={value}
        onChange={(event) =>
          onChange(event.target.value)
        }
        placeholder={placeholder}
        autoComplete={autoComplete}
        required
        className="w-full rounded-xl border border-gray-200 py-3 pl-11 pr-12 text-sm text-gray-900 outline-none transition placeholder:text-gray-400 focus:border-emerald-500 focus:ring-4 focus:ring-emerald-500/10"
      />

      <button
        type="button"
        onClick={onToggle}
        aria-label={
          visible ? 'Hide password' : 'Show password'
        }
        className="absolute right-4 top-1/2 -translate-y-1/2 text-gray-400 transition hover:text-gray-700"
      >
        {visible ? (
          <EyeOff size={18} />
        ) : (
          <Eye size={18} />
        )}
      </button>
    </div>
  )
}
